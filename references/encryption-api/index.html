    Page Encryption Module¶

    Cryptographic primitives, the at-rest envelope format, build-time page helpers, the git filter protocol, and git repository operations behind page encryption and the mkdocs-note crypt commands.

    Called from plugin core during the build and from the CLI for crypt init / status / check / uninstall and the hidden filter commands.

    Cryptographic primitives for mkdocs-note page encryption.

    Two independent layers are used:

    at rest Encrypts the body of a note before it enters git. The password is stored in the same file's clear frontmatter, so a fast HKDF is deliberate: a slow KDF would add build cost without protecting anything the reader cannot already see.

    site Encrypts the rendered page for the browser. When a private repository publishes a public site the frontmatter is not public, so the slow PBKDF2-HMAC-SHA256 stretch is what protects the password.

    Both layers use AES-256-GCM with a synthetic nonce derived through HMAC. A synthetic nonce keeps encryption deterministic, which gives stable git status output and reproducible builds; the AAD binds a payload to its page or key field.

    def normalize_password(value: Any) -> str:
	"""Normalize a password the same way Python and JavaScript do.

	A Unicode NFC form plus surrounding-whitespace stripping keeps a password
	typed in a browser equal to the YAML value even when the source contains

    def password_for_field(meta: Mapping[str, Any], field: str) -> str:
	"""Read the password stored in an explicit frontmatter *field*.

	Used for envelopes, whose header records which field was used when the
	ciphertext was created. Raises :class:`PasswordError` when the field is
	missing or empty.

	if field not in PASSWORD_FIELDS:
		raise PasswordError(
			f"Unsupported encryption key field {field!r}; expected one of "
			f"{', '.join(PASSWORD_FIELDS)}."
		)

	if value is None:
		raise PasswordError(
			f"Frontmatter field {field!r} is missing; it is the password for "
			"this encrypted note."

		raise PasswordError(
			f"Frontmatter field {field!r} is empty; it is the password for this "
			"encrypted note."
		)
	_warn_non_string(field, value)
	return password

    def resolve_password(meta: Mapping[str, Any]) -> PasswordSource:
	"""Resolve a page's password: ``password`` first, then ``permalink``.

	Raises:
		PasswordError: when neither field is present and non-empty.
	"""
	for field in PASSWORD_FIELDS:
		value = meta.get(field)
		if value is None:
			continue
		password = normalize_password(value)
		if not password:
			continue
		_warn_non_string(field, value)
		return PasswordSource(field=field, password=password)
	raise PasswordError(
		"encrypt: true requires a non-empty 'password' or 'permalink' in the "
		"frontmatter."
	)

    def seal_at_rest(password: str, key_field: str, plaintext: bytes) -> bytes:
	"""Encrypt a note body for storage in git.

	Returns ``nonce || ciphertext || tag``. The encryption is deterministic for
	the same ``(password, key_field, plaintext)`` triple.

	ikm = password.encode("utf-8")

	return _seal(enc_key, nonce_key, plaintext, at_rest_aad(key_field))

    def site_salt(src_uri: str) -> bytes:
	"""Stable per-page PBKDF2 salt.

	The salt only has to separate pages, not to be secret; deriving it from the
	stable ``src_uri`` keeps a remembered browser key valid across deploys.

    At-rest envelope format: clear frontmatter plus an encrypted body.

    An envelope keeps the note's YAML frontmatter readable so MkDocs, the plugin scanner, GitHub and git diff can work without a key, and encrypts only the body::

        ---
    title: Example
    permalink: example
    encrypt: true
    ---

    > 🔒 This note is encrypted by mkdocs-note. Read it on the published site.

    <!-- mkdocs-note:encrypted v1 key=permalink
    <base64 of nonce | ciphertext | tag, wrapped at 76 columns>
    -->

    The format is deterministic and fixed per version, which keeps git status stable and makes a byte-exact smudge possible.

    def build_envelope(text: str, key_field: str, password: str) -> str:
	"""Encrypt the body of *text* and return the envelope representation."""
	split = split_frontmatter(text)
	if split is None:
		raise EnvelopeError(
			"Cannot encrypt a note without YAML frontmatter delimited by '---'."
		)
	frontmatter, body, _meta = split
	payload = keys.seal_at_rest(password, key_field, body.encode("utf-8"))
	encoded = base64.b64encode(payload).decode("ascii")
	wrapped = "\n".join(textwrap.wrap(encoded, WRAP_COLUMNS))
	return (
		f"{frontmatter}\n{NOTICE}\n\n"
		f"<!-- {MARKER} v{FORMAT_VERSION} key={key_field}\n{wrapped}\n-->\n"
	)

    def clean_bytes(data: bytes) -> bytes:
	"""Git clean filter: plaintext note bytes -> envelope bytes.

	Files that are not flagged (or are not valid UTF-8) pass through
	byte-for-byte. Flagged files are normalized to UTF-8 without BOM and with LF
	line endings so the envelope is stable across platforms.
	"""
	try:
		text = data.decode("utf-8")
	except UnicodeDecodeError:
		return data
	normalized = normalize_text(text)
	split = split_frontmatter(normalized)
	if split is None:
		return data
	_frontmatter, body, meta = split
	if not keys.is_encrypted(meta):
		return data
	if _marker_in_body(body) is not None:
		# Already an envelope; clean is idempotent.
		return normalized.encode("utf-8")
	source = keys.resolve_password(meta)
	envelope = build_envelope(normalized, source.field, source.password)
	return envelope.encode("utf-8")

    def decrypt_body(
	body: str,
	meta: dict[str, Any],
	password: str | None = None,

	"""Decrypt the body of a page when MkDocs already removed its frontmatter.

    def decrypt_envelope(text: str, password: str | None = None) -> str:
	"""Decrypt a whole-file envelope.

	When *password* is omitted it is read from the frontmatter field recorded in
	the envelope header.
	"""

    def parse_envelope(text: str) -> Envelope | None:
	"""Parse an envelope, or return ``None`` when *text* is not one.

	Raises:
		EnvelopeError: when the marker exists but the payload or version is
			invalid.
	"""
	split = split_frontmatter(text)
	if split is None:
		return None
	frontmatter, body, meta = split
	match = _marker_in_body(body)
	if match is None:
		return None
	version = int(match.group("version"))
	if version != FORMAT_VERSION:
		raise EnvelopeError(
			f"Unsupported encrypted envelope version {version}; "
			f"this mkdocs-note understands v{FORMAT_VERSION}."
		)
	return Envelope(
		frontmatter=frontmatter,
		meta=meta,
		key_field=match.group("field"),
		payload=_decode_payload(match),
		body=body,
	)

    def smudge_bytes(data: bytes) -> bytes:
	"""Git smudge filter: envelope bytes -> plaintext note bytes.

	A checkout must never fail because of a password problem, so an
	undecryptable envelope is emitted unchanged with a single stderr warning.
	"""
	try:
		text = data.decode("utf-8")
	except UnicodeDecodeError:
		return data
	normalized = normalize_text(text)
	try:
		envelope = parse_envelope(normalized)
	except EnvelopeError as exc:
		_warn_smudge(exc)
		return data
	if envelope is None:
		return data
	try:
		password = keys.password_for_field(envelope.meta, envelope.key_field)
		plaintext = keys.open_at_rest(password, envelope.key_field, envelope.payload)
		body = plaintext.decode("utf-8")
	except (keys.CryptoError, UnicodeDecodeError) as exc:
		_warn_smudge(exc)
		return data
	return (envelope.frontmatter + body).encode("utf-8")

    class PacketReader:
	"""Read pkt-line packets from a binary stream."""

	def __init__(self, stream: BinaryIO) -> None:
		self._stream = stream

	def read_packet(self) -> bytes | None:
		"""Read one packet; ``None`` means a flush/delimiter packet.

		Raises:
			EOFError: at end of stream.
			FilterProtocolError: on a malformed length header.
		"""
		header = self._stream.read(4)
		if not header:
			raise EOFError
		if len(header) != 4:
			raise FilterProtocolError("truncated pkt-line header")

			return None

		data = self._stream.read(size - 4)
		if len(data) != size - 4:
			raise FilterProtocolError("truncated pkt-line payload")
		return data

	Args:
		reader: Packet source (git's stdout end).
		writer: Packet sink (git's stdin end).
		stderr: Diagnostic stream (defaults to ``sys.stderr``).
		clean: Plaintext -> envelope transform.
		smudge: Envelope -> plaintext transform.

	Returns:
		Process exit status.

	try:
		_handshake(reader, writer, frozenset(handlers))
	except EOFError:
		return 0
	except FilterProtocolError as exc:
		print(f"mkdocs-note: filter handshake failed: {exc}", file=err)
		return 1

		try:
			request = _read_request(reader)
		except EOFError:
			return 0
		except FilterProtocolError as exc:
			print(f"mkdocs-note: filter protocol error: {exc}", file=err)
			return 1
		if request is None:
			continue

		if handler is None:
			message = f"unsupported filter command {request.command!r}"
			print(f"mkdocs-note: {message}", file=err)
			_write_error(writer, message)
			continue

			output = handler(request.data)
		except Exception as exc:  # noqa: BLE001 - report every failure to git
			prefix = f"{request.pathname}: " if request.pathname else ""
			message = f"{prefix}{exc}"
			print(f"mkdocs-note: {message}", file=err)
			_write_error(writer, message)
			continue

		writer.write_packet(b"status=success")
		writer.write_flush()
		writer.write_content(output)
		writer.write_flush()
		# An empty second status list keeps "status=success" unchanged.
		writer.write_flush()
		writer.flush()

    def check_staged(repo: Repository) -> CheckReport:
	"""Verify that staged encrypted notes are valid envelopes."""
	prefix = f"{repo.docs_rel}/" if repo.docs_rel else ""
	checked: list[str] = []
	errors: list[str] = []

			continue

		except UnicodeDecodeError:
			errors.append(f"{rel}: the staged copy is not valid UTF-8")
			continue

			except keys.PasswordError as exc:
				errors.append(f"{rel}: {exc}")
				continue

					f"{rel}: staged as plaintext. Run 'mkdocs-note crypt init', "
					f"then 'git add --renormalize -- {rel}'."

			except keys.CryptoError as exc:

				f"has 'encrypt: true'; re-add it with 'git add -- {rel}'."
			)

    def notes_with_plaintext_index(repo: Repository) -> list[str]:
	"""Return flagged notes whose staged copy is still plaintext."""
	result: list[str] = []
	for rel, _meta, _text in flagged_notes(repo):
		blob = gitops.blob_bytes(repo.repo_root, rel, ":")
		if blob is None:
			continue
		try:
			staged = blob.decode("utf-8")
		except UnicodeDecodeError:
			result.append(rel)
			continue
		if not envelope.looks_encrypted(staged):
			result.append(rel)
	return result

    def status(repo: Repository) -> StatusReport:
	"""Inspect the filter configuration and every flagged note."""
	commands = {
		key: gitops.config_get(repo.repo_root, key) for key in gitops.FILTER_CONFIG_KEYS
	}
	required = commands.get(gitops.REQUIRED_KEY) == "true"
	hook_path = gitops.git_path(repo.repo_root, "hooks") / "pre-commit"
	try:
		hook_text = hook_path.read_text(encoding="utf-8", errors="replace")
	except OSError:
		hook_text = ""
	hook_installed = HOOK_SIGNATURE in hook_text

	notes: list[NoteStatus] = []
	for rel, meta, text in flagged_notes(repo):
		worktree, work_error = _text_state(text)
		index, index_error = _blob_state(gitops.blob_bytes(repo.repo_root, rel, ":"))
		head, head_error = _blob_state(gitops.blob_bytes(repo.repo_root, rel, "HEAD"))
		errors: list[str] = []
		if work_error:
			errors.append(f"worktree: {work_error}")
		if index in (PLAINTEXT, CORRUPT):
			detail = f" ({index_error})" if index_error else ""
			errors.append(
				f"index: {index}{detail}; re-add with 'git add --renormalize -- {rel}'"
			)
		if head in (PLAINTEXT, CORRUPT):
			detail = f" ({head_error})" if head_error else ""
			errors.append(
				f"HEAD: {head}{detail}; plaintext already committed, rewrite history "
				"if it matters"
			)
		notes.append(
			NoteStatus(
				rel=rel,
				key_field=_key_field(meta),
				worktree=worktree,
				index=index,
				head=head,
				errors=tuple(errors),
			)
		)

	problems: list[str] = []
	if commands.get(gitops.PROCESS_KEY) is None:
		problems.append(
			"filter.mkdocs-note.process is not configured; run 'mkdocs-note crypt init'"
		)
	if not required:
		problems.append(
			"filter.mkdocs-note.required is not true; a missing filter would "
			"silently store plaintext"
		)
	sample = (
		notes[0].rel
		if notes
		else (f"{repo.docs_rel}/index.md" if repo.docs_rel else "index.md")
	)
	attribute = gitops.check_attr(repo.repo_root, sample)
	if attribute != gitops.FILTER_NAME:
		problems.append(
			f".gitattributes does not set 'filter={gitops.FILTER_NAME}' for "
			f"{sample}; run 'mkdocs-note crypt init' and commit .gitattributes"
		)
	for note in notes:
		problems.extend(f"{note.rel}: {error}" for error in note.errors)

	return StatusReport(
		commands=commands,
		required=required,
		attribute=attribute,
		attribute_sample=sample,
		hook_installed=hook_installed,
		notes=tuple(notes),
		problems=tuple(problems),
	)

    def uninstall(repo: Repository) -> UninstallReport:
	"""Remove the filter configuration and the hook installed by ``init``."""
	removed: list[str] = []
	for key in gitops.FILTER_CONFIG_KEYS:
		if gitops.config_unset(repo.repo_root, key):
			removed.append(key)

	hook_path = gitops.git_path(repo.repo_root, "hooks") / "pre-commit"
	hook_removed = False
	hook_kept = False
	if hook_path.is_file():
		try:
			text = hook_path.read_text(encoding="utf-8", errors="replace")
		except OSError:
			text = ""
		if HOOK_SIGNATURE in text:
			try:
				hook_path.unlink()
				hook_removed = True
			except OSError:
				hook_kept = True
		else:
			hook_kept = True

	return UninstallReport(
		removed_keys=tuple(removed),
		hook_removed=hook_removed,
		hook_kept=hook_kept,
	)
